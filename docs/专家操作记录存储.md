  方案设计：专家学习 + 对话历史检索

  一、整体思路

  两个功能都归到一个新的 ExpertLearningUseCase，职责清晰，不侵入已有 UseCase：

  PUT /api/fault-trees/<id>
          │
          ▼
  FaultTreeUseCase.get_by_id()  # 先拿修改前快照
      │
      ▼
  FaultTreeUseCase.update()     # 执行专家修改，得到修改后快照
          │
      └─► ExpertLearningUseCase.learn_from_fault_tree_async(
         tree_id,
         before_tree,
         after_tree,
         update_payload,
           )
                 [后台线程，不阻塞响应]
         ① 将“修改前故障树 + 修改后故障树 + 变更摘要 + 场景上下文”拼成 prompt（promote）
         ② LLM 生成“专家修改学习报告”（强调改了什么、为何修改、适用条件）
         ③ 将学习报告（含前后快照与差异摘要）写入 rag_docs（向量库）
         ④ 抽取三元组并触发一次知识图谱构建（add_triples + save）
         ⑤ 可选：同步写入 graph_entities / graph_relations 向量集合

  ChatUseCase.execute()  [done 事件发送后]
          │
          └─► ExpertLearningUseCase.index_conversation_round(conversation_id, question, answer)
                 [后台线程]
                 ① 拼接 Q&A 文本
                 ② 存入 rag_docs（向量库）

    V2 关键改动（本次更新）：

    - `learn_from_fault_tree_async` 入参升级为：
        `tree_id, before_tree, after_tree, update_payload`
    - 专家学习 prompt（promote）升级为“差异学习模板”：必须包含修改前后故障树与变更摘要。
    - 向量库写入文本不再只存诊断结论，而是存“修改学习记录”（前后快照 + 差异摘要 + 专家解释）。
    - 专家修改后显式触发一次知识图谱构建：`triple_extractor.extract -> graph.add_triples -> graph.save`。
    - 若向量仓库实现支持，则额外写入 `graph_entities / graph_relations` 集合，便于图谱语义检索。

  Backend/Application/UseCases/ExpertLearningUseCase.py

  import logging
  import threading
  from langchain_core.documents import Document
  from Backend.Application.Interfaces.ILLMService import ILLMService
  from Backend.Application.Interfaces.IVectorStoreRepository import IVectorStoreRepository
  from Backend.Application.Interfaces.IGraphRepository import IGraphRepository
  from Backend.Application.Interfaces.ITripleExtractor import ITripleExtractor
  from Backend.Application.Interfaces.IFaultTreeRepository import IFaultTreeRepository
  from Backend.Application.Interfaces.IConversationRepository import IConversationRepository

  logger = logging.getLogger(__name__)

  EXPERT_DIAGNOSIS_SYSTEM =
  "你是一位工业设备故障诊断专家，请根据给定的故障树结构生成一份专业的诊断报告。只输出报告正文，不要附加说明。"

  EXPERT_DIAGNOSIS_USER = """以下是一棵故障树，请生成一份专家诊断报告，包含：
  1. 故障根因分析（结合逻辑门关系说明各层级故障的传播路径）
  2. 关键风险节点识别
  3. 排查建议（按优先级排序）

  故障树名称：{name}

  节点列表：
  {nodes}

  连接关系：
  {edges}

  {conversation_context}
  """


  class ExpertLearningUseCase:
      def __init__(
          self,
          llm_service: ILLMService,
          vector_store_repository: IVectorStoreRepository,
          graph_repository: IGraphRepository,
          triple_extractor: ITripleExtractor,
          fault_tree_repository: IFaultTreeRepository,
          conversation_repository: IConversationRepository,
      ):
          self._llm = llm_service
          self._vector_store = vector_store_repository
          self._graph = graph_repository
          self._triple_extractor = triple_extractor
          self._fault_tree_repo = fault_tree_repository
          self._conversation_repo = conversation_repository

      # ── 故障树学习 ──────────────────────────────────────────

      def learn_from_fault_tree_async(self, tree_id: str) -> None:
          """PUT 接口调用后触发，后台异步执行，不阻塞响应。"""
          t = threading.Thread(target=self._learn_from_fault_tree, args=(tree_id,), daemon=True)
          t.start()

      def _learn_from_fault_tree(self, tree_id: str) -> None:
          try:
              tree = self._fault_tree_repo.get_by_id(tree_id)
              if not tree:
                  logger.warning("ExpertLearning: 找不到故障树 %s", tree_id)
                  return

              # 拼接节点描述
              nodes_text = "\n".join(
                  f"- [{n.node_type}] {n.label}"
                  + (f"（逻辑门: {n.gate_type}）" if n.gate_type else "")
                  + (f" 备注: {n.remark}" if n.remark else "")
                  for n in tree.nodes
              )
              edges_text = "\n".join(
                  f"- {e.source_id} → {e.target_id}" for e in tree.edges
              )

              # 可选：拼入最近 3 轮对话上下文
              conversation_context = ""
              if tree.conversation_id:
                  conv = self._conversation_repo.get_by_id(tree.conversation_id)
                  if conv and conv.rounds:
                      recent = conv.rounds[-3:]
                      qa_text = "\n".join(
                          f"Q: {r.question}\nA: {r.answer}" for r in recent
                      )
                      conversation_context = f"\n参考对话上下文：\n{qa_text}"

              prompt = EXPERT_DIAGNOSIS_USER.format(
                  name=tree.name,
                  nodes=nodes_text,
                  edges=edges_text,
                  conversation_context=conversation_context,
              )
              messages = [
                  {"role": "system", "content": EXPERT_DIAGNOSIS_SYSTEM},
                  {"role": "user", "content": prompt},
              ]

              # 调用 LLM（非流式）
              expert_text = ""
              for token in self._llm.stream_chat(messages):
                  expert_text += token

              if not expert_text.strip():
                  logger.warning("ExpertLearning: LLM 返回空内容，跳过 tree_id=%s", tree_id)
                  return

              source_name = f"expert_diagnosis_{tree_id}"

              # 写入向量库
              doc = Document(
                  page_content=expert_text,
                  metadata={"file_name": source_name, "type": "expert_diagnosis"},
              )
              self._vector_store.add_documents(source_name, [doc])
              logger.info("ExpertLearning: 诊断文本已写入向量库，source=%s", source_name)

              # 抽取三元组写入图谱
              triples = self._triple_extractor.extract(expert_text, source_file=source_name)
              if triples:
                  self._graph.add_triples(triples)
                  self._graph.save()
                  logger.info("ExpertLearning: 写入 %d 条三元组到图谱", len(triples))

          except Exception:
              logger.exception("ExpertLearning: learn_from_fault_tree 失败，tree_id=%s", tree_id)

      # ── 对话历史索引 ────────────────────────────────────────

      def index_conversation_round(
          self, conversation_id: str, question: str, answer: str
      ) -> None:
          """每轮对话结束后调用，后台异步写入向量库。"""
          t = threading.Thread(
              target=self._index_round,
              args=(conversation_id, question, answer),
              daemon=True,
          )
          t.start()

      def _index_round(self, conversation_id: str, question: str, answer: str) -> None:
          try:
              source_name = f"conversation_{conversation_id}"
              content = f"问题：{question}\n\n回答：{answer}"
              doc = Document(
                  page_content=content,
                  metadata={"file_name": source_name, "type": "conversation_history"},
              )
              self._vector_store.add_documents(source_name, [doc])
              logger.info("ExpertLearning: 对话轮次已写入向量库，conversation_id=%s", conversation_id)
          except Exception:
              logger.exception(
                  "ExpertLearning: index_conversation_round 失败，conversation_id=%s",
                  conversation_id,
              )
